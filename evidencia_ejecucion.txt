CREATE TABLE vendedores ( id INTEGER PRIMARY KEY, nombre TEXT NOT NULL, ventas_totales INTEGER NOT NULL CHECK (ventas_totales >= 0), zona TEXT NOT NULL, fecha_ingreso TEXT NOT NULL );
